   115→	public static final void setLogLevel(String levelStr, Class<?>... classes) {

   127→	private static void loadTagProbSource(final TagProbDictionary dictionary, final File file) {
   128→		BufferedReader reader = null;
   129→		try {
   130→			reader = new BufferedReader(new FileReader(file));
   131→			for (String line; (line = reader.readLine()) != null;) {
   132→				dictionary.addSourceEntry(line);
   133→			}
   134→		} catch (final Exception e) {
   135→			logger.error("", e);
   136→		} finally {
   137→			try { reader.close(); } catch (final Exception ignore) { }
   138→		}
   139→	}

   141→	private static void loadTagProbByFileName (TagProbDictionary dictionary, File file) {
   142→		BufferedReader reader = null;
   143→		try {
   144→			Set<CharSequence> entrySet = new HashSet<>();
   145→			String fileName = file.getName();
   146→			String[] parts = fileName.split("\\.");
   147→			String posName = parts[1];
   148→			String probType = parts[2];
   149→			PosTag posTag = null;
   150→			try {
   151→				posTag = PosTag.valueOf(posName);
   152→			} catch (Exception e) {
   153→				logger.error("Undefined pos tag = {}", posName);
   154→				throw e;
   155→			}
   156→			double prob = TagProb.getProb(probType);
   157→
   158→			reader = new BufferedReader(new FileReader(file));
   159→			for (String line; (line = reader.readLine()) != null;) {
   160→				if (line.startsWith("#") || line.startsWith("//") || line.length() == 0) { continue; }
   161→				CharVector cv = new CharVector(line);
   162→				if (dictionary.ignoreCase()) { cv.ignoreCase(); }
   163→				entrySet.add(cv);
   164→			}
   165→			dictionary.appendPosTagEntry(entrySet, posTag, prob);
   166→		} catch (final Exception e) {
   167→			logger.error("", e);
   168→		} finally {
   169→			try { reader.close(); } catch (final Exception ignore) { }
   170→		}
   171→	}

   176→	public static final ProductNameDictionary loadExtraDictionary(ProductNameDictionary dictionary) {
   177→		File dictDir = new File(RESOURCE_ROOT, "dict_extra");

   179→
   180→		/*사용자 사전*/
   181→		SetDictionary userDict = new SetDictionary(true);
   182→		userDict.loadSource(new File(dictDir, "user.txt"));
   183→		dictionary.addDictionary(ProductNameDictionary.DICT_USER, userDict);
   184→		dictionary.appendAdditionalNounEntry(userDict.set(), HIGH);
   185→		/*유사어 사전*/
   186→		SynonymDictionary synonymDict = new SynonymDictionary(true);
   187→		synonymDict.loadSource(new File(dictDir, "synonym.txt"));
   188→		dictionary.addDictionary(ProductNameDictionary.DICT_SYNONYM, synonymDict);
   189→		dictionary.appendAdditionalNounEntry(synonymDict.getWordSet(), HIGH);
   190→		/*단위명 사전*/
   191→		SetDictionary unitDict = new SetDictionary(true);
   192→		unitDict.loadSource(new File(dictDir, "unit.txt"));
   193→		dictionary.addDictionary(ProductNameDictionary.DICT_UNIT, unitDict);
   194→		/*단위명유사어 사전*/
   195→		SynonymDictionary unitSynDict = new SynonymDictionary(true);
   196→		unitSynDict.loadSource(new File(dictDir, "unit_synonym.txt"));
   197→		dictionary.addDictionary(ProductNameDictionary.DICT_UNIT_SYNONYM, unitSynDict);
   198→		/*분리어 사전*/
   199→		SpaceDictionary spaceDict = new SpaceDictionary(true);
   200→		spaceDict.loadSource(new File(dictDir, "space.txt"));
   201→		dictionary.addDictionary(ProductNameDictionary.DICT_SPACE, spaceDict);
   202→		/*복합명사 사전*/
   203→		CompoundDictionary compDict = new CompoundDictionary(true);
   204→		compDict.loadSource(new File(dictDir, "compound.txt"));
   205→		dictionary.addDictionary(ProductNameDictionary.DICT_COMPOUND, compDict);
   206→		return dictionary;
   207→	}

   227→	public static final ProductNameDictionary loadDictionary() {
   228→		ProductNameDictionary dictionary = null;
   229→		String path = getSystemProperty(SYSPROP_TEST_DICTIONARY_SETTING);
   230→		File propFile = null;
   231→		int loadType = 0;
   232→		/** 
   233→		 * 테스트 실행시 사전 읽어오기
   234→		 *   InteliJ 에서 실행시 :
   235→		 *     Modify Run Configuration -> VM options 에 다음과 같이 추가한다
   236→		 *     -DSYSPROP_TEST_DICTIONARY_SETTING=C:/{{플러그인경로}}/product-name-dictionary.yml
   237→		 * 
   238→		 *   Eclipse 에서 실행시 :
   239→		 *     Run -> Run Configurations -> Arguments -> VM arguments 에 다음과 같이 추가한다.
   240→		 *     -DSYSPROP_TEST_DICTIONARY_SETTING=C:/{{플러그인경로}}/product-name-dictionary.yml
   241→		 *     
   242→		 *   VSCode 에서 실행시 :
   243→		 *     .vscode/settings.json 파일 에서 다음과 같이 추가한다.
   244→		 *     {
   245→		 *       "java.test.config": {
   246→		 *         "vmargs": [
   247→		 *           -DSYSPROP_TEST_DICTIONARY_SETTING=C:/{{플러그인경로}}/product-name-dictionary.yml
   248→		 *         ]
   249→		 *       }
   250→		 *     }
   251→		 * 
   252→		 * 1. 지정하지 않은경우 : 내부적으로 컴파일된 사전(src/test/resources/product-name-dictionary.yml) 읽어옴
   253→		 *   -DSYSPROP_TEST_DICTIONARY_SETTING
   254→		 * 
   255→		 * 2. 직접지정한 경우 : 외부 사전 읽어옴
   256→		 *   -DSYSPROP_TEST_DICTIONARY_SETTING=C:/Temp/analysis-product/product-name-dictionary.yml
   257→		 * 
   258→		 * 3. INTERNAL 로 지정한 경우 : 텍스트베이스로 작성된 사전 읽어옴
   259→		 *   -DSYSPROP_TEST_DICTIONARY_SETTING=INTERNAL
   260→		 * 
   261→		 **/
   262→		if (INTERNAL.equals(path)) {
   263→			/* 3. INTERNAL TEXT BASED DICTIONARY */
   264→			loadType = 1;
   265→		} else if (path != null && !"".equals(path)) {
   266→			/* 2. EXTERNAL DICTIONARY */
   267→			propFile = TestUtil.getFileByPath(path, null);
   268→			loadType = 0;
   269→		} else {
   270→			/* 1. INTERNAL COMPILED BINARY */
   271→			path = new File(RESOURCE_ROOT, "product-name-dictionary.yml").getAbsolutePath();
   272→			propFile = TestUtil.getFileByPath(path, null);
   273→			loadType = 0;
   274→		}
   275→		try {
   276→			if (loadType == 0) {
   277→				JSONObject prop = TestUtil.readYmlConfig(propFile);
   278→				dictionary = ProductNameDictionary.loadDictionary(propFile.getParentFile(), prop);
   279→			} else {
   280→				dictionary = loadInternalDictionary();
   281→			}
   282→		} catch (Exception e) {
   283→			logger.debug("ERROR LOADING DICTIONARY : {}", e.getMessage());
   284→		}
   285→		/**
   286→		 * 테스트 사전에서 추가적인 단어 적재가 필요 없을경우 다음과 같이 지정
   287→		 * -DSYSPROP_TEST_DICTIONARY_LOAD_EXTRA=NONE
   288→		 **/
   289→		if (!NONE.equals(getSystemProperty(SYSPROP_TEST_DICTIONARY_LOAD_EXTRA))) {
   290→			TestUtil.loadExtraDictionary(dictionary);
   291→		}
   292→		return dictionary;
   293→	}
