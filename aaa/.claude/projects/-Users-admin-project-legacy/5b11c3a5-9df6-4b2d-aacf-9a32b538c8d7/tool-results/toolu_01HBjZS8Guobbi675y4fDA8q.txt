    64→    public String getDirectoryAllFiles(String path) {
    65→        StringBuffer sb = new StringBuffer();
    66→
    67→        try {
    68→            if (Files.isDirectory(Paths.get(path))) {
    69→                logger.info("{} is directory !", path);
    70→                Files.walk(Paths.get(path))
    71→                        .filter(Files::isRegularFile)
    72→                        .forEach(item -> {
    73→                            if (sb.length() == 0) {
    74→                                sb.append(item.toString());
    75→                            } else {
    76→                                sb.append("," + item.toString());
    77→                            }
    78→                        });
    79→            } else {
    80→                logger.info("{} is regularFile !", path);
    81→                sb.append(path);
    82→            }
    83→        } catch (IOException e) {
    84→            e.printStackTrace();
    85→        }
    86→
    87→        return sb.toString();
    88→    }
