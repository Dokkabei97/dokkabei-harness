+@Service
+class LegacyReserveSyncApplicationService(
+    private val legacyReserveSyncRepository: LegacyReserveSyncRepository,
+    private val cachedProxyApplicationService: CachedProxyApplicationService,
+) {
+    private val logger by logger()
+
+    fun syncToLegacy(changedData: List<ChangedData<LegacyProductData>>) {
+        if (changedData.isEmpty()) return
+
+        val dependencyId = getDependencyId(changedData)
+        val dependency = cachedProxyApplicationService.fetchDependencies(dependencyId)
+
+        runBlocking(Dispatchers.IO) {
+            // push 태스크 수집 → supervisorScope로 병렬 실행 (개별 실패 격리)
+            val pushTasks = buildPushTasks(changedData, dependency.categories, dependency.malls)
+
+            if (pushTasks.isNotEmpty()) {
+                supervisorScope {
+                    pushTasks
+                        .map { task ->
+                            async {
+                                runCatching { task() }
+                                    .onFailure { ex -> logger.error(ex) { "Legacy push 실패" } }
+                            }
+                        }.awaitAll()
+                }
+            }
+        }
+    }
+
+    private fun buildPushTasks(
+        changedData: List<ChangedData<LegacyProductData>>,
+        categories: Map<Int, CategoryProjection>,
+        malls: Map<Int, MallProjection>,
+    ): List<suspend () -> Unit> {
+        val pushTasks = mutableListOf<suspend () -> Unit>()
+        val grouped = changedData.groupBy { it.op }
+
+        grouped.forEach { (op, items) ->
+            val linked = mutableListOf<LinkedProductRequest>()
+            val unlinked = mutableListOf<UnlinkedProductRequest>()
+            val unlinkedDeletes = mutableListOf<UnlinkedProductDeleteRequest>()
+
+            items.forEach items@{ item ->
+                val data = if (op == Operation.DELETE) item.beforeData else item.afterData
+                if (data == null) {
+                    logger.warn { "Legacy sync skipped: no data for product id=${item.id}, op=$op" }
+                    return@items
+                }
+
+                when {
+                    data.isCatalogMatch -> {
+                        linked.add(LegacySyncMapper.convertToLinkedRequest(data, malls.getValue(data.shopId)))
+                    }
+
+                    op == Operation.DELETE -> {
+                        unlinkedDeletes.add(LegacySyncMapper.convertToUnlinkedDeleteRequest(data, malls.getValue(data.shopId)))
+                    }
+
+                    else -> {
+                        unlinked.add(
+                            LegacySyncMapper.convertToUnlinkedRequest(
+                                data,
+                                categories.getValue(data.categoryId),
+                                malls.getValue(data.shopId),
+                            )
+                        )
+                    }
+                }
+            }
+
+            if (linked.isNotEmpty()) pushTasks.add { legacyReserveSyncRepository.pushLinkedProducts(linked, op) }
+            if (unlinked.isNotEmpty()) pushTasks.add { legacyReserveSyncRepository.pushUnlinkedProducts(unlinked, op) }
+            if (unlinkedDeletes.isNotEmpty()) {
+                pushTasks.add {
+                    legacyReserveSyncRepository.pushUnlinkedProductDeletes(
+                        unlinkedDeletes,
+                    )
+                }
+            }
+        }
+
+        return pushTasks
+    }
+
+    fun getDependencyId(documents: List<ChangedData<LegacyProductData>>): DependencyId {
+        val categoryIds: MutableList<Int> = mutableListOf()
+        val mallIds: MutableList<Int> = mutableListOf()
+
+        documents.forEach {
+            it.afterData?.categoryId?.run { categoryIds.add(this) }
+            it.afterData?.shopId?.run { mallIds.add(this) }
+        }
+
+        return DependencyId(
+            categoryIds = categoryIds.distinct().toSet(),
+            mallIds = mallIds.distinct().toSet(),
+        )
+    }

+# CLAUDE.md
+
+This file provides guidance to Claude Code (claude.ai/code) when working with code in this repository.
+
+## 서비스 개요
+
+**legacy-sync**는 Kafka CDC 상품 이벤트를 소비하여 레거시 백엔드 시스템에 HTTP API로 동기화하는 Spring Boot 서비스이다. dynamic-indexer가 Elasticsearch에 색인하는 것과 달리, legacy-sync는 상품 데이터를 **연동상품(PDM)**과 **비연동상품(TCMPNY_LINK)** 두 경로로 분류하여 레거시 API의 `/pushQueue.json` 엔드포인트에 전송한다.
+
+## Commands
+
+```bash
+# 빌드
+./gradlew :bootstrap:legacy-sync:build
+
+# 로컬 실행
+./gradlew :bootstrap:legacy-sync:bootRun --args='--spring.profiles.active=local'
+
+# 의존 모듈별 테스트
+./gradlew :domain:index:test
+./gradlew :application:index:test
+./gradlew :outbound-adapter:legacy:test
+./gradlew :inbound-adapter:legacy:test
+
+# 단일 테스트 클래스 실행
+./gradlew :outbound-adapter:legacy:test --tests "com.cowave.search.outbound_adapter.legacy.LegacySyncRepositoryImplTest"
+
+# 코드 포맷팅 (커밋 전 필수)
+./gradlew ktlintFormat
+
+# 로컬 인프라 (ES, Kafka, Valkey, Schema Registry 등)
+docker compose up -d
+```

+
+```
+bootstrap/legacy-sync               ← DI 조립, Spring Boot 진입점

+## 데이터 플로우
+
+```
+Kafka (Product CDC 토픽, Avro)
+    │
+    ▼
+LegacyReserveSyncListener (배치 소비)
+    │  filter: categoryId > 0
+    │  map: Avro → ChangedData<Product>
+    ▼
+ProductMergeStrategy.merge()
+    │  같은 ID 이벤트 병합 (offset 순 정렬 → fold)
+    ▼
+LegacySyncMapper.convertToChangedLegacyData()
+    │  Product → LegacyProductData 변환
+    │  DANAWA 채널 데이터 추출 (price, url, benefit, promotion)
+    ▼
+LegacyReserveSyncApplicationService.syncToLegacy()
+    │
+    ├─ getDependencyId() → categoryIds, mallIds 추출
+    ├─ CachedProxyApplicationService.fetchDependencies() → 캐시에서 조회
+    │
+    └─ buildPushTasks() → Operation별 그룹화 후 상품 분류:
+        │
+        ├─ isCatalogMatch == true  →  LinkedProductRequest    → pushQueue("PDM", POST/PUT/DELETE)
+        ├─ op == DELETE            →  UnlinkedDeleteRequest   → pushQueue("TCMPNY_LINK", DELETE)
+        └─ else (CREATE/UPDATE)    →  UnlinkedProductRequest  → pushQueue("TCMPNY_LINK", POST/PUT)
+                                                                    │
+                                                                    ▼
+                                                            Legacy API
+                                                     /pushQueue.json?service={service}
+```
+
+## 핵심 설계
+
+### 상품 분류 전략
+
+| 조건 | 분류 | 대상 서비스 | HTTP 메서드 |
+|------|------|------------|------------|
+| `isCatalogMatch == true` | 연동상품 | PDM | POST/PUT/DELETE |
+| `isCatalogMatch == false` + DELETE | 비연동 삭제 | TCMPNY_LINK | DELETE |
+| `isCatalogMatch == false` + CREATE/UPDATE | 비연동상품 | TCMPNY_LINK | POST/PUT |
+
+카탈로그 매칭 판정: `catalogMatchings?.any { it.source != "NONE" }`
+
+### 조건부 활성화
+
+`@ConditionalOnProperty(name = ["legacy-sync.enabled"], havingValue = "true")`로 Bean이 등록된다. `application.yml`에서 `index.enabled: false`, `dynamic-indexer.enabled: false`를 설정하여 다른 서비스 컴포넌트와 충돌을 방지한다.
+
+### 비동기 실행 & 장애 격리
+
+- Kafka 리스너는 blocking, 내부 HTTP 호출은 `runBlocking(Dispatchers.IO)` + `supervisorScope`
+- `supervisorScope` 내에서 pushTask별 `async` 실행 → 하나의 실패가 다른 push를 차단하지 않음
+- 리스너 레벨 예외 시 Kafka ACK 미수행 → `DefaultErrorHandler`가 1초 간격 무한 재시도
+
+### 의존성 캐싱
+
+Category, Mall 데이터는 `CachedProxyApplicationService`를 통해 Valkey 캐시에서 조회한다. 배치 내 모든 상품의 고유 ID를 추출하여 한 번에 조회하므로 네트워크 호출을 최소화한다. TTL은 600초(10분).
+
+## Kafka 컨슈머 설정
+
+| 항목 | 값 |
+|------|-----|
+| 토픽 | `cmn.pricecompare-be-team.cdc.mongo.shopping.products.v1` (dynamic-indexer와 동일) |
+| Group ID | `cmn-search-team-group-legacy-sync` (별도 컨슈머 그룹) |
+| Concurrency | 3 |
+| AckMode | `MANUAL_IMMEDIATE` |
+| 에러 핸들링 | `DefaultErrorHandler` (1초 간격, 무한 재시도) |
+| Serializer | Avro (KafkaAvroDeserializer) |
+
+프로퍼티 바인딩: `LegacyReserveSyncConsumerProperties` (`@ConfigurationProperties(prefix = "legacy-sync-consumer")`)
+
+## Legacy API 클라이언트
+
+- **엔드포인트**: `/pushQueue.json?service={PDM|TCMPNY_LINK}`
+- **WebFlux 기반**: `WebClient` + Reactor Netty, 코루틴 `awaitSingle()`
+- **커넥션 풀**: max 50, timeout 500ms (connect + response)
+- **프로퍼티**: `LegacyApiClientProperties` (`@ConfigurationProperties(prefix = "client.legacy-api")`)
+
+## OutputPort 구조
+
+```
+application/index/repository/LegacyReserveSyncRepository.kt   ← 인터페이스 (OutputPort)
+outbound-adapter/legacy/LegacyReserveSyncRepositoryImpl.kt     ← 구현체 (HTTP 클라이언트 래핑)
+```
+
+세 가지 suspend 메서드:
+- `pushLinkedProducts(products, operation)` → PDM 서비스
+- `pushUnlinkedProducts(products, operation)` → TCMPNY_LINK 서비스
+- `pushUnlinkedProductDeletes(products)` → TCMPNY_LINK DELETE

+## dynamic-indexer와의 차이
+
+| 구분 | dynamic-indexer | legacy-sync |
+|------|-----------------|-------------|
+| 데이터 싱크 | Elasticsearch (bulk API) | Legacy HTTP API (/pushQueue.json) |
+| 처리 대상 | Catalog + Product CDC | Product CDC만 |
+| 상품 분류 | 단일 인덱스 | 연동(PDM) / 비연동(TCMPNY_LINK) |
+| HTTP 메서드 | ES bulk (PUT/POST) | 명시적 POST/PUT/DELETE |
+| 타임아웃 | bulk 크기에 따라 가변 | 500ms 고정 |
+| offset 관리 | ConsumerCallbackManager (seek/rewind) | 단순 manual ACK |

+        acknowledgment: Acknowledgment?,
+        consumer: Consumer<*, *>,
+    ) {
+        runCatching {
+            val changedDocuments =
+                data.mapNotNull { record ->
+                    val envelope = record.value() as ProductCdcEnvelope
+                    val product = envelope.after ?: envelope.before
+
+                    if (product == null) {

+                        null
+                    } else if (product.categoryId == null || product.categoryId == 0) {

+                        null
+                    } else {
+                        record.toChangedProductData()
+                    }
+                }
+
+            val mergedDocuments = ProductMergeStrategy.merge(changedDocuments)
+            legacyReserveSyncApplicationService.syncToLegacy(LegacySyncMapper.convertToChangedLegacyData(mergedDocuments))
+        }.onSuccess {
+            acknowledgment?.acknowledge()
+        }.onFailure { ex ->
+            logger.error(ex) { "Failed to sync products to legacy" }

+package com.cowave.search.client.legacy_api

+import com.cowave.shared.utils.logger
+import kotlinx.coroutines.reactor.awaitSingle
+import org.springframework.beans.factory.annotation.Qualifier
+import org.springframework.http.HttpMethod
+import org.springframework.http.MediaType
+import org.springframework.stereotype.Component
+import org.springframework.web.reactive.function.client.WebClient
+import org.springframework.web.reactive.function.client.bodyToMono
+
+@Component
+class LegacyApiClient(
+    @param:Qualifier("LegacyWebClient")
+    private val client: WebClient,
+) {
+    private val logger by logger()
+
+    suspend fun pushQueue(
+        service: String,
+        method: HttpMethod,
+        body: Any,
+    ) {
+        try {
+            client
+                .method(method)
+                .uri("/pushQueue.json?service={service}", service)
+                .contentType(MediaType.APPLICATION_JSON)
+                .bodyValue(body)
+                .retrieve()
+                .bodyToMono<String>()
+                .awaitSingle()
+        } catch (ex: Exception) {
+            logger.error { "Legacy API pushQueue 호출 실패 [service=$service]: $ex" }
+            throw ex
+        }
+    }
