+import org.springframework.stereotype.Service
+
+@Service
+@ConditionalOnProperty(name = ["legacy-sync.enabled"], havingValue = "true", matchIfMissing = true)
+class LegacySyncApplicationService(
+    private val legacySyncRepository: LegacySyncRepository,
+    private val categoryService: CategoryService,
+    private val mallService: MallService,
+) {
+    private val logger by logger()
+
+    fun syncToLegacy(changedData: List<ChangedData<LegacyProductData>>) {
+        if (changedData.isEmpty()) return
+
+        // beforeData도 포함하여 DELETE 시에도 mall/category 조회 가능하도록 수정
+        val categoryIds =
+            changedData
+                .flatMap { listOfNotNull(it.afterData?.categoryId, it.beforeData?.categoryId) }
+                .distinct()
+        val mallIds =
+            changedData
+                .flatMap { listOfNotNull(it.afterData?.shopId, it.beforeData?.shopId) }
+                .distinct()
+
+        // TODO: 캐시를 먼저 확인하게 수정 필요 !!
+        val categories = if (categoryIds.isNotEmpty()) categoryService.getCategories(categoryIds) else emptyList()
+        val malls = if (mallIds.isNotEmpty()) mallService.getMalls(mallIds).associateBy { it.mallId } else emptyMap()
+
+        val grouped = changedData.groupBy { it.op }
+
+        grouped.forEach { (op, items) ->
+            val linked = mutableListOf<LinkedProductRequest>()
+            val unlinked = mutableListOf<UnlinkedProductRequest>()
+            val unlinkedDeletes = mutableListOf<UnlinkedProductDeleteRequest>()
+
+            items.forEach items@{ item ->
+                val data = if (op == Operation.DELETE) item.beforeData else item.afterData
+                if (data == null) {
+                    logger.warn { "Legacy sync skipped: no data for product id=${item.id}, op=$op" }
+                    return@items
+                }
+
+                val mall = malls[data.shopId]
+
+                when {
+                    // 매칭상품 (PDM) - CREATE/UPDATE/DELETE 모두
+                    data.isCatalogMatch -> {
+                        linked.add(LegacySyncMapper.toLinkedRequest(data, mall))
+                    }
+
+                    // 미매칭 DELETE - 간소화 요청
+                    op == Operation.DELETE -> {
+                        unlinkedDeletes.add(LegacySyncMapper.toUnlinkedDeleteRequest(data, mall))
+                    }
+
+                    // 미매칭 CREATE/UPDATE - 전체 필드
+                    else -> {
+                        val category = categories[data.categoryId]
+                        unlinked.add(LegacySyncMapper.toUnlinkedRequest(data, category, mall))
+                    }
+                }
+            }
+
+            if (linked.isNotEmpty()) legacySyncRepository.pushLinkedProducts(linked, op)
+            if (unlinked.isNotEmpty()) legacySyncRepository.pushUnlinkedProducts(unlinked, op)
+            if (unlinkedDeletes.isNotEmpty()) legacySyncRepository.pushUnlinkedProductDeletes(unlinkedDeletes)
+        }
+    }

+package com.cowave.search.repository.cache
+
+import com.cowave.shared.serialization.defaultJson
+import kotlinx.serialization.KSerializer
+import org.springframework.data.redis.core.RedisTemplate
+
+/**
+ * `ValkeyRepository`는 valkey를 DB로써 사용하는 영속성 레이어의 리포지토리입니다.
+ */
+abstract class AbstractValkeyRepository<ID, V : Any> (
+    private val redisTemplate: RedisTemplate<String, String>,
+    private val serializer: KSerializer<V>,
+    private val keyPrefix: String,
+) {
+
+    private fun serialize(value: V): String = defaultJson.encodeToString(serializer, value)
+
+    private fun deserialize(jsonString: String): V = defaultJson.decodeFromString(serializer, jsonString)
+
+    /**
+     * Missed keys에 대한 처리를 애플리케이션에서 주도적으로 할 수 있도록 반환에 null이 포함될 수 있습니다.
+     */
+    protected fun multiGet(ids: List<ID>): List<V?> {
+        return redisTemplate.opsForValue()
+            .multiGet(ids.toKeys())
+            ?.map { it?.let(::deserialize) }
+            ?: emptyList()
+    }
+
+    protected fun hashMultiGet(ids: List<ID>): List<V?> {
+        return redisTemplate.opsForHash<String, String>()
+            .multiGet(getHashKey(), ids.toFields())
+            .map { it?.let(::deserialize) }
+    }
+
+    /**
+     * Valkey는 DB로써 사용되므로 TTL을 가지고 있지 않어 mset이 적합합니다.
+     */
+    protected fun multiSet(keyValuePairs: Map<ID, V>) {
+        redisTemplate.opsForValue()
+            .multiSet(
+                keyValuePairs.map { (k, v) ->
+                    k.toKey() to serialize(v)
+                }.toMap()
+            )
+    }
+
+    protected fun hashMultiSet(keyValuePairs: Map<ID, V>) {
+        return redisTemplate.opsForHash<String, String>()
+            .putAll(
+                getHashKey(),
+                keyValuePairs.map { (k, v) ->
+                    k.toField() to serialize(v)
+                }.toMap()
+            )
+    }
+
+    protected fun getHashKey(): String = keyPrefix
+
+    protected fun ID.toKey(): String = "$keyPrefix:$this"
+
+    protected fun List<ID>.toKeys(): List<String> = this.map { it.toKey() }
+
+    protected fun ID.toField(): String = this.toString()
+
+    protected fun List<ID>.toFields(): List<String> = this.map { it.toField() }
+}

+package com.cowave.shared.cache
+
+import com.github.benmanes.caffeine.cache.Cache
+
+class DefaultLocalCache<ID : Any, V : Any>(
+    val cache: Cache<String, V>,
+    val keyPrefix: String,
+) : LocalCache<ID, V> {
+
+    override fun get(id: ID): V? =
+        cache.getIfPresent(id.toKey())
+
+    override fun getAll(ids: Collection<ID>): Map<ID, V?> {
+        val values: List<V?> = ids.map { id -> id.toKey() }
+            .map { key -> cache.getIfPresent(key) }
+
+        return ids.zip(values).associate {
+            it.first to it.second
+        }
+    }
+
+    override fun put(id: ID, value: V) {
+        cache.put(id.toKey(), value)
+    }
+
+    override fun putAll(values: Map<ID, V>) {
+        values
+            .map {it.key.toKey() to it.value }
+            .toMap()
+            .let { cache.putAll(it) }
+    }
+
+    private fun ID.toKey(): String = "$keyPrefix:id"
+}

+/**
+ * 각 cache, fetcher 마다 사용하는 key의 형태가 다르기 때문에 공통의 id를 key로써 사용하고 캐시의 실제 키는 각 캐시 내부에서 처리합니다.
+ */
+class DefaultTieredLoadingCache<ID, V>(
+    private val localCache: LocalCache<ID, V>,
+    private val globalCache: GlobalCache<ID, V>,
+    private val fetcher: (Collection<ID>) -> Map<ID, V>,
+) : TieredLoadingCache<ID, V> {
+
+    override fun get(id: ID): V? {
+        // 1. L1
+        localCache.get(id)?.let { return it }
+
+        // 2. L2
+        val globalCached = globalCache.getAll(listOf(id))
+        globalCached[id]?.let {
+            localCache.put(id, it)
+            return it
+        }
+
+        // 3. L3
+        val fetched = fetcher(listOf(id))
+        fetched[id]?.let {
+            localCache.put(id, it)
+            globalCache.putAll(mapOf(id to it))
+            return it
+        }
+
+        // 4. All missed
+        return null
+    }
+
+    override fun getAll(ids: Collection<ID>): Map<ID, V?> {
+        if (ids.isEmpty()) return emptyMap()
+
+        // 1. L1
+        val localCached = localCache.getAll(ids)
+        val localMiss = localCached.filterValues { it == null }
+        if (localMiss.keys.isEmpty()) return localCached
+
+        // 2. L2
+        val globalCached = globalCache.getAll(localMiss.keys)
+        val globalHit = globalCached.filterValues { it != null }.mapValues { it.value!! }
+        val globalMiss = globalCached.filterValues { it == null }
+        if (globalHit.isNotEmpty()) {
+            localCache.putAll(globalHit)
+        }
+        if (globalMiss.isEmpty()) return localCached + globalCached
+
+        // 3. L3
+        val fetched = fetcher(globalMiss.keys)
+        if (fetched.isNotEmpty()) {
+            localCache.putAll(fetched)
+            globalCache.putAll(fetched)
+        }
+
+        return localCached + globalCached + fetched
+    }
