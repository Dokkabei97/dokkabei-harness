    74→        private fun getDictionaryFile(envBase: File?, prop: JSONObject, basePath: String?): File {
    75→            var localBaseFile: File? = null

    77→                if (!basePath.isNullOrEmpty()) {

    81→                    localBaseFile = envBase
    82→                }

    84→                logger.debug("DICTIONARY EXCEPTION : {} / {}", localBaseFile, e.message)
    85→                localBaseFile = envBase

    88→            val dictionaryId = prop.optString(ATTR_DICTIONARY_NAME, "").trim()

    92→                ret = File(File(envBase, DICTIONARY_PATH), "$dictionaryId$DICTIONARY_SUFFIX")

   170→        @JvmStatic
   171→        fun loadDictionary(baseFile: File?, dictProp: JSONObject?): ProductNameDictionary? {
   172→            if (dictProp == null) return null
   173→
   174→            var dictionary: Dictionary<TagProb, PreResult<CharSequence>>? = null
   175→            var commonDictionary: ProductNameDictionary? = null
   176→            val dictList = dictProp.optJSONArray(ATTR_DICTIONARY_LIST) ?: return null
   177→            val basePath = dictProp.optString(ATTR_DICTIONARY_BASE_PATH)
   178→
   179→            // 시스템사전을 먼저 읽어오도록 한다.
   180→            for (inx in 0 until dictList.length()) {
   181→                val row = dictList.optJSONObject(inx)
   182→                if (getType(row) == DictionaryType.SYSTEM) {
   183→                    dictionary = loadSystemDictionary(baseFile!!, row, basePath)
   184→                    commonDictionary = ProductNameDictionary(dictionary)
   185→                    break
   186→                }
   187→            }
   188→
   189→            if (commonDictionary == null) return null
   190→
   191→            for (inx in 0 until dictList.length()) {
   192→                val row = dictList.optJSONObject(inx)
   193→                val dictionaryId = row.optString(ATTR_DICTIONARY_NAME)
   194→                val type = getType(row) ?: continue
   195→                val tokenType = getTokenType(row)
   196→                val label = getLabel(row)
   197→                val seq = getSeq(row)
   198→                val dictFile = getDictionaryFile(baseFile, row, basePath)
   199→                val ignoreCase = getIgnoreCase(row)
   200→                var sourceDictionary: SourceDictionary<*>? = null
   201→                logger.info("Dictionary Setting Name: {}, type: {}", label, type.name)
   202→
   203→                when (type) {
   204→                    DictionaryType.SET -> {
   205→                        val setDictionary = SetDictionary(dictFile, ignoreCase, label, seq, tokenType, type)
   206→                        if (tokenType != null) {
   207→                            commonDictionary.appendAdditionalNounEntry(setDictionary.set(), tokenType)
   208→                        }
   209→                        sourceDictionary = setDictionary
   210→                    }
   211→                    DictionaryType.MAP -> {
   212→                        val mapDictionary = MapDictionary(dictFile, ignoreCase, label, seq, tokenType, type)
   213→                        if (tokenType != null) {
   214→                            commonDictionary.appendAdditionalNounEntry(mapDictionary.map().keys, tokenType)
   215→                        }
   216→                        sourceDictionary = mapDictionary
   217→                    }
   218→                    DictionaryType.SYNONYM, DictionaryType.SYNONYM_2WAY -> {
   219→                        val synonymDictionary = SynonymDictionary(dictFile, ignoreCase, label, seq, tokenType, type)
   220→                        if (tokenType != null) {
   221→                            commonDictionary.appendAdditionalNounEntry(synonymDictionary.getWordSet(), tokenType)
   222→                        }
   223→                        sourceDictionary = synonymDictionary
   224→                    }
   225→                    DictionaryType.SPACE -> {
   226→                        val spaceDictionary = SpaceDictionary(dictFile, ignoreCase, label, seq, tokenType, type)
   227→                        if (tokenType != null) {
   228→                            commonDictionary.appendAdditionalNounEntry(spaceDictionary.getWordSet(), tokenType)
   229→                        }
   230→                        sourceDictionary = spaceDictionary
   231→                    }
   232→                    DictionaryType.CUSTOM -> {
   233→                        val customDictionary = CustomDictionary(dictFile, ignoreCase, label, seq, tokenType, type)
   234→                        if (tokenType != null) {
   235→                            commonDictionary.appendAdditionalNounEntry(customDictionary.getWordSet(), tokenType)
   236→                        }
   237→                        sourceDictionary = customDictionary
   238→                    }
   239→                    DictionaryType.INVERT_MAP -> {
   240→                        val invertMapDictionary = InvertMapDictionary(dictFile, ignoreCase, label, seq, tokenType, type)
   241→                        if (tokenType != null) {
   242→                            commonDictionary.appendAdditionalNounEntry(invertMapDictionary.map().keys, tokenType)
   243→                        }
   244→                        sourceDictionary = invertMapDictionary
   245→                    }
   246→                    DictionaryType.COMPOUND -> {
   247→                        val compoundDictionary = CompoundDictionary(dictFile, ignoreCase, label, seq, tokenType, type)
   248→                        if (tokenType != null) {
   249→                            commonDictionary.appendAdditionalNounEntry(compoundDictionary.map().keys, tokenType)
   250→                        }
   251→                        sourceDictionary = compoundDictionary
   252→                    }
   253→                    DictionaryType.SYSTEM -> { /* ignore */ }
   254→                }
   255→
   256→                if (sourceDictionary != null) {
   257→                    commonDictionary.addDictionary(dictionaryId, sourceDictionary)
   258→                }
   259→            }
   260→            return commonDictionary
   261→        }

   263→        @JvmStatic
   264→        fun reloadDictionary() {
   265→            val localBaseFile = baseFile
   266→            val localConfigFile = configFile
   267→            if (localBaseFile == null || localConfigFile == null) {
   268→                logger.error("DICTIONARY NOT LOADED!")
   269→                return
   270→            }
   271→            val newCommonDictionary = loadDictionary(localBaseFile, ResourceResolver.readYmlConfig(localConfigFile))
   272→            val commonDictionary = contextStore.getAs(PRODUCT_NAME_DICTIONARY, ProductNameDictionary::class.java)
   273→                ?: return
   274→
   275→            // 1. commonDictionary에 systemdictionary셋팅.
   276→            newCommonDictionary?.let { commonDictionary.reset(it) }
   277→            // 2. dictionaryMap 에 셋팅.
   278→            val dictionaryMap = commonDictionary.getDictionaryMap()
   279→
   280→            for ((dictionaryId, dictionary) in dictionaryMap) {
   281→                when {
   282→                    dictionary.javaClass.isAssignableFrom(SetDictionary::class.java) -> {
   283→                        val setDictionary = dictionary as SetDictionary
   284→                        val newDictionary = newCommonDictionary?.getDictionary(dictionaryId, SetDictionary::class.java)
   285→                        newDictionary?.let { setDictionary.setSet(it.set()) }
   286→                    }
   287→                    dictionary.javaClass.isAssignableFrom(MapDictionary::class.java) -> {
   288→                        val mapDictionary = dictionary as MapDictionary
   289→                        val newDictionary = newCommonDictionary?.getDictionary(dictionaryId, MapDictionary::class.java)
   290→                        newDictionary?.let { mapDictionary.setMap(it.map()) }
   291→                    }
   292→                    dictionary.javaClass.isAssignableFrom(SynonymDictionary::class.java) -> {
   293→                        val synonymDictionary = dictionary as SynonymDictionary
   294→                        val newDictionary = newCommonDictionary?.getDictionary(dictionaryId, SynonymDictionary::class.java)
   295→                        newDictionary?.let {
   296→                            synonymDictionary.setMap(it.map())
   297→                            synonymDictionary.setWordSet(it.getWordSet())
   298→                        }
   299→                    }
   300→                    dictionary.javaClass.isAssignableFrom(SpaceDictionary::class.java) -> {
   301→                        val spaceDictionary = dictionary as SpaceDictionary
   302→                        val newDictionary = newCommonDictionary?.getDictionary(dictionaryId, SpaceDictionary::class.java)
   303→                        newDictionary?.let {
   304→                            spaceDictionary.setMap(it.map())
   305→                            spaceDictionary.setWordSet(it.getWordSet())
   306→                        }
   307→                    }
   308→                    dictionary.javaClass.isAssignableFrom(CustomDictionary::class.java) -> {
   309→                        val customDictionary = dictionary as CustomDictionary
   310→                        val newDictionary = newCommonDictionary?.getDictionary(dictionaryId, CustomDictionary::class.java)
   311→                        newDictionary?.let {
   312→                            customDictionary.setMap(it.map())
   313→                            customDictionary.setWordSet(it.getWordSet())
   314→                        }
   315→                    }
   316→                    dictionary.javaClass.isAssignableFrom(CompoundDictionary::class.java) -> {
   317→                        val compoundDictionary = dictionary as CompoundDictionary
   318→                        val newDictionary = newCommonDictionary?.getDictionary(dictionaryId, CompoundDictionary::class.java)
   319→                        newDictionary?.let { compoundDictionary.setMap(it.map()) }
   320→                    }
   321→                }
   322→                logger.info("Dictionary [{}] is reloaded!", dictionaryId)
   323→            }
   324→        }

   326→        @JvmStatic
   327→        fun compileDictionary(
   328→            client: NodeClient,
   329→            repo: DictionaryRepository,
   330→            filter: String?,
   331→            exportFile: Boolean
   332→        ) {
   333→            val localBaseFile = baseFile
   334→            val localConfigFile = configFile
   335→            if (localBaseFile == null || localConfigFile == null) {
   336→                logger.error("DICTIONARY NOT LOADED!")
   337→                return
   338→            }
   339→
   340→            val filters = mutableSetOf<String>()
   341→            filter?.trim()?.split(",")?.forEach { filters.add(it.lowercase()) }
   342→            logger.info("Compile Dict types: {}", filters)
   343→
   344→            val dictProp = ResourceResolver.readYmlConfig(localConfigFile) ?: return
   345→            val dictList = dictProp.optJSONArray(ATTR_DICTIONARY_LIST) ?: return
   346→            val basePath = dictProp.optString(ATTR_DICTIONARY_BASE_PATH)
   347→
   348→            for (inx in 0 until dictList.length()) {
   349→                val row = dictList.optJSONObject(inx)
   350→                val dictionaryId = row.optString(ATTR_DICTIONARY_NAME)
   351→
   352→                // 필터가 존재하면서 dictionaryId가 없다면 스킵!
   353→                if (filters.isNotEmpty() && !filters.contains(dictionaryId)) {
   354→                    continue
   355→                }
   356→
   357→                // 사전과 컴파일된 Count를 Map형태로 받는다.
   358→                val compileDictMap = compileDictionaryOne(repo, dictList, dictionaryId) ?: continue
   359→                @Suppress("UNCHECKED_CAST")
   360→                val sourceDictionary = compileDictMap["dict"] as? SourceDictionary<*>
   361→                if (sourceDictionary != null) {
   362→                    // 엘라스틱서치에 .dsearch_dict_apply 인덱스에 upsert 한다.
   363→                    DictionaryUpsertService.upsertData(client, ".dsearch_dict_apply", compileDictMap)
   364→
   365→                    // 파일로 기록. 컴파일.
   366→                    if (exportFile) {
   367→                        val dictFile = getDictionaryFile(localBaseFile, row, basePath)
   368→                        try {
   369→                            FileOutputStream(dictFile).use { ostream ->
   370→                                sourceDictionary.writeTo(ostream)
   371→                                logger.info("DICTIONARY FILE SAVED! {} / {}", dictionaryId, dictFile.absolutePath)
   372→                            }
   373→                        } catch (ignore: Exception) {
   374→                        }
   375→                    }
   376→                }
   377→            }
   378→            logger.debug("DICTIONARY LOAD COMPLETE! [{}]", filter ?: "ALL")
   379→        }

   381→        @JvmStatic
   382→        fun compileDictionaryOne(
   383→            repo: DictionaryRepository,
   384→            dictList: JSONArray,
   385→            dictionaryId: String
   386→        ): MutableMap<String, Any?>? {
   387→            val result = mutableMapOf<String, Any?>()
   388→
   389→            // dictList 설정을 스캔하면서 dictionaryId를 찾는다.
   390→            for (inx in 0 until dictList.length()) {
   391→                val row = dictList.optJSONObject(inx)
   392→                val dictId = row.optString(ATTR_DICTIONARY_NAME)
   393→                if (dictionaryId != dictId) continue
   394→
   395→                val type = getType(row) ?: continue
   396→                val tokenType = getTokenType(row)
   397→                val label = getLabel(row)
   398→                val seq = getSeq(row)
   399→                val ignoreCase = row.optBoolean(ATTR_DICTIONARY_IGNORECASE, true)
   400→
   401→                val source = repo.getSource(dictionaryId)
   402→
   403→                // 사전 타입으로 객체생성
   404→                val sourceDictionary: SourceDictionary<*>? = when (type) {
   405→                    DictionaryType.SET -> SetDictionary(ignoreCase)
   406→                    DictionaryType.MAP -> MapDictionary(ignoreCase)
   407→                    DictionaryType.SYNONYM, DictionaryType.SYNONYM_2WAY -> SynonymDictionary(ignoreCase)
   408→                    DictionaryType.SPACE -> SpaceDictionary(ignoreCase)
   409→                    DictionaryType.CUSTOM -> CustomDictionary(ignoreCase)
   410→                    DictionaryType.INVERT_MAP -> InvertMapDictionary(ignoreCase)
   411→                    DictionaryType.COMPOUND -> CompoundDictionary(ignoreCase)
   412→                    else -> null
   413→                }
   414→
   415→                // 사전객체에 데이터 입력
   416→                if (sourceDictionary != null) {
   417→                    var cnt = 0
   418→                    while (source.hasNext()) {
   419→                        val data = source.next()
   420→                        val id = data[0]?.toString()?.trim() ?: ""
   421→                        val keyword = data[1]?.toString()?.trim() ?: ""
   422→                        val value = data[2]?.toString()?.trim() ?: ""
   423→
   424→                        val line = when (type) {
   425→                            DictionaryType.SYNONYM, DictionaryType.SYNONYM_2WAY -> {
   426→                                if (keyword.isNotEmpty()) "$keyword\t$value" else value
   427→                            }
   428→                            DictionaryType.CUSTOM -> {
   429→                                if (id.isNotEmpty()) "$keyword\t$id" else keyword
   430→                            }
   431→                            else -> {
   432→                                if (value.isNotEmpty()) "$keyword\t$value" else keyword
   433→                            }
   434→                        }
   435→                        sourceDictionary.addSourceLineEntry(line)
   436→                        cnt++
   437→                    }
   438→                    result["cnt"] = cnt
   439→                    logger.info("DICTIONARY LOADED! [{}] / {} / {} / {}", cnt, dictionaryId, type, tokenType)
   440→                }
   441→                result["dict"] = sourceDictionary
   442→                result["type"] = dictId
   443→                result["id"] = dictId
   444→                return result
   445→            }
   446→            return null
   447→        }

   449→        @JvmStatic
   450→        fun getTwowaySynonymWord(
   451→            mainWord: CharSequence,
   452→            synonymMap: Map<CharSequence, Array<CharSequence>>
   453→        ): List<CharSequence>? {
   454→            val map = mutableMapOf<CharSequence, MutableList<CharSequence>>()
   455→            val keys = mutableListOf<CharSequence>()
   456→
   457→            val initialValues = mutableListOf<CharSequence>()
   458→            map[mainWord] = initialValues
   459→            keys.addAll(map.keys)
   460→            synonymMap[mainWord]?.let { initialValues.addAll(it) }
   461→
   462→            var kinx = 0
   463→            while (kinx < keys.size) {
   464→                val word = keys[kinx]
   465→                val synonyms = synonymMap[word]
   466→                logger.trace("SYN {} : {}", word, synonyms)
   467→
   468→                if (synonyms != null) {
   469→                    for (synonym in synonyms) {
   470→                        if (keys.contains(synonym)) continue
   471→                        val subSynonyms = synonymMap[synonym]
   472→                        if (subSynonyms != null && synonyms.isNotEmpty()) {
   473→                            for (subSynonym in subSynonyms) {
   474→                                if (mainWord == subSynonym) {
   475→                                    val newValues = mutableListOf<CharSequence>()
   476→                                    map[synonym] = newValues
   477→                                    keys.add(synonym)
   478→                                    newValues.addAll(subSynonyms)
   479→                                    break
   480→                                }
   481→                            }
   482→                        }
   483→                    }
   484→                }
   485→                kinx++
   486→            }
   487→
   488→            if (map.size > 1) {
   489→                normalizeSynonymMap(map)
   490→            }
   491→            logger.trace("MAP:{}", map)
   492→
   493→            return if (map.size > 1) {
   494→                keys.clear()
   495→                keys.addAll(TreeSet(map.keys))
   496→                keys
   497→            } else {
   498→                null
   499→            }
   500→        }

   502→        @JvmStatic
   503→        fun normalizeSynonymMap(map: MutableMap<CharSequence, MutableList<CharSequence>>) {
   504→            val keys = mutableListOf<CharSequence>()
   505→            keys.addAll(map.keys)
   506→
   507→            // 키값에 없는 동의어 제거
   508→            for (word in keys) {
   509→                val values = map[word] ?: continue
   510→                val iterator = values.iterator()
   511→                while (iterator.hasNext()) {
   512→                    val value = iterator.next()
   513→                    if (!keys.contains(value)) {
   514→                        iterator.remove()
   515→                    }
   516→                }
   517→            }
   518→
   519→            val delete = mutableListOf<CharSequence>()
   520→            // 동의어에 없는 키값 제거
   521→            logger.trace("MAP:{}", map)
   522→            for (sinx in keys.indices) {
   523→                val word = keys[sinx]
   524→                for (tinx in keys.indices) {
   525→                    if (sinx == tinx) continue
   526→                    val values = map[keys[tinx]]
   527→                    if (values != null && !values.contains(word)) {
   528→                        delete.add(word)
   529→                        break
   530→                    }
   531→                }
   532→            }
   533→            for (key in delete) {
   534→                map.remove(key)
   535→            }
   536→            logger.trace("MAP:{}", map)
   537→        }

   539→        @JvmStatic
   540→        fun isOneWaySynonym(word: CharSequence, map: Map<CharSequence, Array<CharSequence>>): Boolean {
   541→            if (map.containsKey(word)) {
   542→                val twoway = getTwowaySynonymWord(word, map)
   543→                logger.trace("TWO-WAY {} = {}", word, twoway)
   544→                val synonyms = map[word] ?: return false
   545→                for (value in synonyms) {
   546→                    if (twoway != null && twoway.contains(value)) continue
   547→                    // 단방향이 하나라도 있으면 단방향 동의어
   548→                    logger.trace("ONE-WAY {} -> {}", word, value)
   549→                    return true
   550→                }
   551→            }
   552→            return false
   553→        }

   559→            val dictionaryMap = productNameDictionary.getDictionaryMap()

   561→            for ((key, sourceDictionary) in dictionaryMap) {
   562→                logger.debug("KEY:{} / {}", key, sourceDictionary)
   563→
   564→                when {
   565→                    sourceDictionary.javaClass.isAssignableFrom(SetDictionary::class.java) -> {
   566→                        val dictionary = sourceDictionary as SetDictionary
   567→                        repo.restore(key, dictionary.ignoreCase(), dictionary.set())
   568→                    }
   569→                    sourceDictionary.javaClass.isAssignableFrom(MapDictionary::class.java) -> {
   570→                        val dictionary = sourceDictionary as MapDictionary

   573→                            val sb = StringBuilder()

   575→                                if (sb.isNotEmpty()) sb.append(",")
   576→                                sb.append(value.toString().trim())
   577→                            }
   578→                            words.add("$word$TAB$sb")
   579→                        }
   580→                        repo.restore(key, dictionary.ignoreCase(), words)
   581→                    }
   582→                    sourceDictionary.javaClass.isAssignableFrom(SynonymDictionary::class.java) -> {
   583→                        val dictionary = sourceDictionary as SynonymDictionary
   584→                        val words = mutableSetOf<CharSequence>()
   585→                        for ((word, values) in dictionary.map()) {
   586→                            val twoway = getTwowaySynonymWord(word, dictionary.map())
   587→                            // 단방향 동의어에서 양방향을 뺀 나머지 값들만 저장
   588→                            val sb = StringBuilder()

   590→                                if (twoway != null && twoway.contains(value)) continue
   591→                                if (sb.isNotEmpty()) sb.append(",")
   592→                                sb.append(value.toString().trim())
   593→                            }

   595→                                words.add("$word$TAB$sb")
   596→                            }
   597→                            // 양방향 동의어 저장
   598→                            if (twoway != null && twoway.isNotEmpty()) {
   599→                                val sbTwoway = StringBuilder()

   601→                                    if (sbTwoway.isNotEmpty()) sbTwoway.append(",")
   602→                                    sbTwoway.append(value.toString().trim())
   603→                                }
   604→                                if (sbTwoway.isNotEmpty()) {
   605→                                    words.add("$TAB$sbTwoway")
   606→                                }
   607→                            }

   611→                    sourceDictionary.javaClass.isAssignableFrom(SpaceDictionary::class.java) -> {
