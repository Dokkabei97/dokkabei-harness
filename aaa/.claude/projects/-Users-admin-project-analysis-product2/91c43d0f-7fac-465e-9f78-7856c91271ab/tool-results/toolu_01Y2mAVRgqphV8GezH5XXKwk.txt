   132→    fun getTermType(cv: CharSequence): String? {
   133→        var prevType: String? = null
   134→        var type: String? = null
   135→        for (inx in 0 until cv.length) {
   136→            prevType = type
   137→            type = getType(cv[inx].code)
   138→            if (prevType != null && type != prevType) {
   139→                type = when {
   140→                    (prevType == ALPHA && type == NUMBER) || (prevType == NUMBER && type == ALPHA) -> ALPHANUM
   141→                    prevType == ALPHANUM && (type == ALPHA || type == NUMBER) -> ALPHANUM
   142→                    (prevType == ALPHA && type == SYMBOL) || (prevType == NUMBER && type == SYMBOL) ||
   143→                    (prevType == ALPHANUM && type == SYMBOL) -> ASCII
   144→                    prevType == ASCII && (type == ALPHA || type == NUMBER || type == SYMBOL) -> ASCII
   145→                    else -> UNCATEGORIZED
   146→                }
   147→            }
   148→        }
   149→        return type
   150→    }
