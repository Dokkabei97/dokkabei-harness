   154→	@Override protected RestChannelConsumer prepareRequest(RestRequest request, NodeClient client) throws IOException {
   155→		final String action = request.param("action");
   156→		final StringWriter buffer = new StringWriter();
   157→		JSONWriter builder = new JSONWriter(buffer);
   158→
   159→		if (ACTION_TEST.equals(action)) {
   160→			testAction(request, client, builder);
   161→			builder.object().key("action").value(action).endObject();
   162→		} else if (ACTION_RELOAD_DICT.equals(action)) {
   163→			reloadDictionary();
   164→			builder.object().key("action").value(action).endObject();
   165→		} else if (ACTION_INFO_DICT.equals(action)) {
   166→			builder.object().key("action").value(action);
   167→			infoDictionary(request, client, builder);
   168→			builder.endObject();
   169→		} else if (ACTION_FIND_DICT.equals(action)) {
   170→			builder.object().key("action").value(action);
   171→			findDictionary(request, client, builder);
   172→			builder.endObject();
   173→		} else if (ACTION_RESTORE_DICT.equals(action)) {
   174→			restoreDictionary(request, client);
   175→			builder.object().key("action").value(action).endObject();
   176→		} else if (ACTION_COMPILE_DICT.equals(action)) {
   177→			compileDictionary(request, client);
   178→			builder.object().key("action").value(action).endObject();
   179→		} else if (ACTION_FULL_INDEX.equals(action)) {
   180→			int count = bulkIndex(request, client);
   181→			builder.object().key("action").value(action);
   182→			if (count > 0) {
   183→				builder.key("working").value("true").key("count").value(count);
   184→			} else {
   185→				builder.key("working").value("false");
   186→			}
   187→			builder.endObject();
   188→		} else if (ACTION_FASTCAT_INDEX.equals(action)) {
   189→			int count = fastcatIndex(request, client);
   190→			builder.object().key("action").value(action);
   191→			if (count > 0) {
   192→				builder.key("working").value("true").key("count").value(count);
   193→			} else {
   194→				builder.key("working").value("false");
   195→			}
   196→			builder.endObject();
   197→		} else if (ACTION_ANALYZE_TEXT.equals(action)) {
   198→			analyzeTextAction(request, client, builder);
   199→		} else if (ACTION_SEARCH.equals(action)) {
   200→			search(request, client, builder);
   201→		} else if (ACTION_BUILD_QUERY.equals(action)) {
   202→			buildQuery(request, client, builder);
   203→		}else if(ACTION_ANALYZE_MULTI_PARAMS.equals(action)){
   204→			builder.object();
   205→			analyzeMultiParamsAction(request, client, builder);
   206→			builder.endObject();
   207→		} else if(ACTION_SYNONYM_LIST.equals(action)){
   208→			builder.object();
   209→			getSynonymListAction(request, client, builder);

   211→		} else if (ACTION_SEARCH_KEYWORD.equals(action)) {

   214→			builder.object().key("loaded").value(getDictionary() != null);
   215→			builder.endObject();
   216→		}
   217→		return channel -> {
   218→			channel.sendResponse(new RestResponse(RestStatus.OK, CONTENT_TYPE_JSON, new BytesArray(buffer.toString().getBytes(StandardCharsets.UTF_8))));
   219→		};
   220→	}

   240→	/**
   241→	 * 각 노드에 신호 전파
   242→	 */
   243→	private void distribute(NodeClient client, String action, JSONObject body, boolean selfDist) {
   244→		String localNodeId = client.getLocalNodeId();
   245→		NodesInfoRequest infoRequest = new NodesInfoRequest();
   246→		infoRequest.clear()
   247→			.addMetrics(NodesInfoMetrics.Metric.JVM.metricName())
   248→			.addMetrics(NodesInfoMetrics.Metric.OS.metricName())
   249→			.addMetrics(NodesInfoMetrics.Metric.PROCESS.metricName())
   250→			.addMetrics(NodesInfoMetrics.Metric.HTTP.metricName())
   251→			.addMetrics(NodesInfoMetrics.Metric.PLUGINS.metricName())
   252→			.addMetrics(NodesInfoMetrics.Metric.INDICES.metricName());
   253→
   254→		try {
   255→			NodesInfoResponse response = client.admin().cluster().nodesInfo(infoRequest).get();
   256→			List<NodeInfo> nodes = response.getNodes();
   257→			for (NodeInfo node : nodes) {
   258→				// 자기자신이 아닌경우에만 전파
   259→				if (!selfDist && localNodeId.equals(node.getNode().getId())) {
   260→					continue;
   261→				}
   262→				boolean hasPlugin = false;
   263→				TransportAddress address = node.getInfo(HttpInfo.class).address().publishAddress();
   264→				// 상품명분석기 플러그인을 가진 노드에만 전파
   265→				List<PluginRuntimeInfo> plugins = node.getInfo(PluginsAndModules.class).getPluginInfos();
   266→				for (PluginRuntimeInfo info : plugins) {
   267→					if (hasPlugin = info.descriptor().getClassname().equals(AnalysisProductNamePlugin.class.getName())) {
   268→						break;
   269→					}
   270→				}
   271→				if (hasPlugin) {
   272→					logger.debug("NODE: {}:{}", address.getAddress(), address.getPort());
   273→					doRestRequest(address.getAddress(), address.getPort(), action, body);
   274→				}
   275→			}
   276→		} catch (Exception e) {
   277→			logger.error("", e);
   278→		}
   279→	}

   281→	/**
   282→	 * 특정 노드에 REST 신호를 보낸다
   283→	 */
   284→	private String doRestRequest(final String address, final int port, final String action, final JSONObject body) {
   285→		StringBuilder sb = new StringBuilder();
   286→		HttpURLConnection con = null;
   287→		OutputStream ostream = null;
   288→		BufferedReader reader = null;
   289→		try {
   290→			String url = "http://" + address + ":" + port + BASE_URI + "/" + action;
   291→			logger.debug("SEND REQUEST {}", url);
   292→			con = (HttpURLConnection) new URL(url).openConnection();
   293→			con.setRequestMethod("POST");
   294→			con.setRequestProperty("Content-Type", "application/json");
   295→			con.setDoOutput(true);
   296→			ostream = con.getOutputStream();
   297→			ostream.write(String.valueOf(body).getBytes());
   298→			ostream.flush();
   299→			int responseCode = con.getResponseCode();
   300→			if (responseCode == HttpURLConnection.HTTP_CREATED) {
   301→				logger.trace("RESPONSE:{}", responseCode);
   302→				reader = new BufferedReader(new InputStreamReader(con.getInputStream()));
   303→				for (String rl; (rl = reader.readLine()) != null;) {
   304→					sb.append(rl).append("\r\n");
   305→				}
   306→				logger.trace("RESPONSE:{}", sb);
   307→			}
   308→		} catch (Exception e) {
   309→			logger.error("", e);
   310→		} finally {
   311→			try { ostream.close(); } catch (Exception ignore) { }
   312→			try { reader.close(); } catch (Exception ignore) { }
   313→		}
   314→		return String.valueOf(sb);
   315→	}

   359→	private void analyzeMultiParamsAction(RestRequest request, NodeClient client, JSONWriter writer) {
   360→		JSONObject jparam = new JSONObject();
   361→
   362→		logger.debug("multi-params Action");
   363→		// 인덱스는 고정
   364→		String index = ES_DICTIONARY_INDEX;
   365→
   366→		// 검색어 확장
   367→		boolean useForQuery = false;
   368→		boolean useSynonym = true;
   369→		boolean useStopword = false;
   370→		boolean useFullString = true;
   371→
   372→		Map<String, String> analyzeMap = new HashMap<>();
   373→		if (!GET.equals(request.method())) {
   374→			// POST 처리
   375→			jparam = parseRequestBody(request);
   376→			for (String key : jparam.keySet()) {
   377→				analyzeMap.put(key, jparam.optString(key, ""));
   378→			}
   379→		} else {
   380→			// GET 처리
   381→			try {
   382→				for (String key : request.params().keySet()) {
   383→					if (key.equals("action") || key.equals("pretty")) continue;
   384→					analyzeMap.put(key, URLDecoder.decode(request.param(key, ""), "UTF-8"));
   385→				}
   386→			} catch (UnsupportedEncodingException e) {
   387→				logger.error("UnsupportedEncodingException Catched !!! >>>> \n" + e.getMessage());
   388→				writer.key("success").value(false);
   389→				logger.debug("error >>> {}", e);
   390→			}
   391→		}
   392→
   393→		TokenStream stream = null;
   394→		writer.key("result").object();
   395→		try {
   396→			for(String key : analyzeMap.keySet()){
   397→				String value = analyzeMap.get(key);
   398→				logger.debug("key : {} :::::: value : {}", key, value);
   399→				stream = ProductNameAnalyzerProvider.getAnalyzer(value, useForQuery, useSynonym, useStopword, useFullString, false);
   400→				List<String> list = analyzeMultiParamsText(client, value, stream, index, useForQuery);
   401→				if(list.size() != 0){
   402→					writer.key(key).array();
   403→					for(String item : list){
   404→						writer.value(item);
   405→					}
   406→					writer.endArray();
   407→				}
   408→				stream.close();
   409→			}
   410→		} catch (Exception ignore){
   411→			writer.endObject();
   412→			writer.key("success").value(false);
   413→			logger.debug("exception >>> {}", ignore);
   414→			logger.error("Exception Catched !!! >>>> \n" + ignore.getMessage());
   415→		} finally {
   416→			writer.endObject();
   417→			writer.key("success").value(true);
   418→		}
   419→	}

   421→	/**
   422→	 * 단방향 동의어 인지 확인
   423→	 */
   424→	public static boolean isOneWaySynonym(NodeClient client, String index, String word) {
   425→		boolean ret = false;
   426→		// 단방향 판단
   427→		if (client != null && index != null && !"".equals(index)) {
   428→			// 색인을 사용하는 경우
   429→			BoolQueryBuilder query = QueryBuilders.boolQuery()
   430→				.must(QueryBuilders.matchQuery(ES_DICT_FIELD_TYPE, ProductNameDictionary.DICT_SYNONYM.toUpperCase()))
   431→				.must(QueryBuilders.matchQuery(ES_DICT_FIELD_KEYWORD, word)
   432→				);
   433→			logger.trace("Q:{}", query);
   434→			long count = 0;
   435→			if ((count = SearchUtil.count(client, index, query)) > 0) {
   436→				ret = true;
   437→			}
   438→			logger.trace("SYNONYM:{}-{} = {}", index, word, count);
   439→		} else {
   440→			// 메모리 사전을 이용하는 경우
   441→			ProductNameDictionary dictionary = getDictionary();
   442→			SynonymDictionary synonyms = dictionary.getDictionary(ProductNameDictionary.DICT_SYNONYM, SynonymDictionary.class);
   443→			ret = ProductNameDictionary.isOneWaySynonym(CharVector.valueOf(word), synonyms.map());
   444→		}
   445→		return ret;
   446→	}

  1808→		CharTermAttribute termAttr = stream.addAttribute(CharTermAttribute.class);
  1809→		TypeAttribute typeAttr = stream.addAttribute(TypeAttribute.class);
  1810→		SynonymAttribute synAttr = stream.addAttribute(SynonymAttribute.class);
  1811→		Map<String, List<List<String>>> result = new HashMap<>();

  1816→		}
  1817→

  1819→		String type = null;
  1820→		String setName = null;
  1821→		try {
  1822→			stream.reset();
  1823→			while (stream.incrementToken()) {
  1824→				term = String.valueOf(termAttr);
  1825→				type = typeAttr.type();
  1826→				logger.trace("TERM:{} / {}", term, type);
  1827→				if (ProductNameTokenizer.FULL_STRING.equals(type)) {
  1828→					// 전체 단어

  1831→				} else if (ProductNameTokenizer.MODEL_NAME.equals(type)) {
  1832→					// 모델명의 경우 색인시와 질의시 추출방법이 서로 다르다.
  1833→					setName = ANALYZE_SET_MODEL_NAME;
  1834→					setAnalyzedResult(result, term, ANALYZE_SET_MODEL_NAME, ANALYZE_SET_FINAL);
  1835→				} else {
  1836→					// 일반단어
  1837→					setName = ANALYZE_SET_NORMAL;
