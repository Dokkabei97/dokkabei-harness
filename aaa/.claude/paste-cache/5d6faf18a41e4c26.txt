# External Secrets용 Source Secret YAML 예시

################################### 경고 ###################################
##                                                                        ##
##  해당 파일에 실제 계정정보 및 비밀번호를 입력해서 git에 올리지 마세요. ##
##  이 파일은 운영전 까지 사용하는 파일입니다.                            ##
##  이 파일을 서버에서 clone/pull 한 후 실제 비밀번호로 변경하세요.       ##
##                                                                        ##
############################################################################
---
apiVersion: v1
kind: Namespace
metadata:
  name: dev-search

---
# MongoDB Credentials
apiVersion: v1
kind: Secret
metadata:
  name: common-mongodb-credentials
  namespace: dev-search
type: Opaque
stringData:
  MONGO_USERNAME: changeme  # ← 실제 MongoDB 사용자명으로 변경
  MONGO_PASSWORD: changeme  # ← 실제 MongoDB 비밀번호로 변경

---
# Elasticsearch Frontend Cluster (API Key)
apiVersion: v1
kind: Secret
metadata:
  name: common-elasticsearch-frontend-credentials
  namespace: dev-search
type: Opaque
stringData:
  ES_FRONTEND_API_KEY: changeme-frontend-api-key  # ← 실제 Frontend ES API Key로 변경

---
# Elasticsearch Backend Cluster (API Key)
apiVersion: v1
kind: Secret
metadata:
  name: common-elasticsearch-backend-credentials
  namespace: dev-search
type: Opaque
stringData:
  ES_BACKEND_API_KEY: changeme-backend-api-key  # ← 실제 Backend ES API Key로 변경

---
# Valkey (Redis) Credentials
apiVersion: v1
kind: Secret
metadata:
  name: common-valkey-credentials
  namespace: dev-search
type: Opaque
stringData:
  VALKEY_PASSWORD: changeme  # ← 실제 Valkey 비밀번호로 변경

---
# Kafka Credentials
apiVersion: v1
kind: Secret
metadata:
  name: common-kafka-credentials
  namespace: dev-search
type: Opaque
stringData:
  KAFKA_USERNAME: changeme  # ← 실제 Kafka 사용자명으로 변경
  KAFKA_PASSWORD: changeme  # ← 실제 Kafka 비밀번호로 변경
