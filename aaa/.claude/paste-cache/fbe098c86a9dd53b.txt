SELECT
        tP.prod_c as id,
        tP.prod_c as productCode,
        'dna' as shopCode,
        tP.prod_n as productName,
        NVL(tPM.maker_n ,'') as productMaker,
        NVL(tKFS.sMakerKeywordList,'') as makerKeyword,
        NVL(tPR.brand_n,'') as productBrand,
        NVL(tKFS.sBrandKeywordList,'') as brandKeyword,
        NVL(SUBSTR((SELECT GROUP_CONCAT(DISTINCT(tSKD.sModelKeyword) , ', ') FROM tSearchKeywordModel tSKD WHERE tP.prod_c = tSKD.prod_c AND tSKD.SKEYWORDTYPE = 'R' GROUP BY tP.prod_c),1,255),'') as productModel,
        NVL(tKFS.sModelKeywordList,'') as modelWeight,
        NVL(tPD.sSimpleDescription,'') as simpleDescription,
        NVL(tPD.sAddDescription,'') as addDescription,        
        NVL(tPD.sCmDescription,'') as cmDescription,
        tPD.sMobileSimpleDescription as mobileDescription,
        tPCDAT.sActionTagCode as actionTag,
        tPCDAT.sMobileActionTagCode as mobileActionTag,
        tP.nLowPrice as pcPrice,
        tP.nLowPrice as lowestPrice,
        CASE
            WHEN tP.NLOWPRICEINCLUDEDELIVERYCOST = 0 THEN tP.nLowPrice
            ELSE tP.NLOWPRICEINCLUDEDELIVERYCOST
        END AS totalPrice,
        tP.avgprice_q as averagePrice,
        tP.nLowPrice as mobilePrice,
        tP.shop_q as shopQuantity,
        NVL(tP.shot_yn ,'N') as discontinued,
        tPC.nCategorySeq1 as categoryCode1,
        tPC.nCategorySeq2 as categoryCode2,
        NVL(tPC.nCategorySeq3, 0) as categoryCode3,
        NVL(tPC.nCategorySeq4, 0) as categoryCode4,        
        CASE
            WHEN tCFS.nCategorySeq3 is null THEN tCFS.sCategoryName1 || '>' || tCFS.sCategoryName2
            WHEN tCFS.nCategorySeq3='' THEN tCFS.sCategoryName1 || '>' || tCFS.sCategoryName2
            WHEN tCFS.nCategorySeq3=0 THEN tCFS.sCategoryName1 || '>' || tCFS.sCategoryName2
            WHEN tCFS.nCategorySeq4 is null THEN tCFS.sCategoryName1 || '>' || tCFS.sCategoryName2 || '>' || tCFS.sCategoryName3
            WHEN tCFS.nCategorySeq4='' THEN tCFS.sCategoryName1 || '>' || tCFS.sCategoryName2 || '>' || tCFS.sCategoryName3
            WHEN tCFS.nCategorySeq4=0 THEN tCFS.sCategoryName1 || '>' || tCFS.sCategoryName2 || '>' || tCFS.sCategoryName3
            ELSE tCFS.sCategoryName1 || '>' || tCFS.sCategoryName2 || '>' || tCFS.sCategoryName3 || '>' || tCFS.sCategoryName4
        END as categoryName,
        tCFS.sCategoryKeywordList as categoryKeyword,
        tCFS.sCategoryWeightList as categoryWeight1,
        (SELECT group_concat(tAVFS.sWeightKeyword, ',')
         FROM tProductAttributeValue tPA
         LEFT JOIN TATTRIBUTEVALUEFORSEARCH tAVFS ON (tAVFS.nAttributeValueSeq = tPA.NATTRIBUTEVALUESEQ)
         WHERE tPA.nProductCategorySeq = tPC.nProductCategorySeq
         GROUP BY tPA.nProductCategorySeq) as categoryWeight2,
        TO_CHAR(tP.input_d, 'YYYYMMDD') as registerDate,
        TO_CHAR(tP.input_d, 'YYYYMMDD HH24:MI:SS') as registerDateTime,
        TO_CHAR(tP.dtPriceModifyDate, 'YYYYMMDD') as modifyDate,
        TO_CHAR(tP.modify_d, 'YYYYMMDD HH24:MI:SS') as modifyDateTime,
        TO_CHAR(tP.make_d, 'YYYYMMDD') as manufactureDate,
        TO_CHAR(tP.make_d, 'YYYYMMDD HH24:MI:SS') as manufactureDateTime,
        CASE
            WHEN tP.shot_yn = 'Y'  THEN 0
            WHEN tP.nLowPrice > 0  THEN tPC.nCntSave + 1
            ELSE
                CASE
                    WHEN tP.dtPriceModifyDate < DATEADD(SYSDATE, -30, 'DAY') THEN tPC.nCntSave + 1
                    ELSE 0
                END
        END AS popularityScore,
        NVL(tPC.nCntSave,0) as cntSave,
        NVL(tKFS.sKeywordList, '') as managerKeyword,
        TO_CHAR(tPSM.MODIFY_DT, 'YYYYMMDD') as managerKeywordDate,
        CASE
            WHEN tP.shot_yn = 'Y'  THEN '1'
            WHEN tP.nLowPrice > 0  THEN '7' 
            ELSE
                CASE
                    WHEN tP.dtPriceModifyDate < DATEADD(SYSDATE, -30,'DAY') THEN
                        CASE
                            WHEN tP.dtPriceModifyDate = '01-JAN-1970' THEN
                                CASE
                                    WHEN tP.make_d > SYSDATE THEN '5'
                                    ELSE '4'
                                END
                            ELSE '3'
                        END
                    ELSE '3'
                END
        END as productClassification,
        tP.nPromotionMinPriceQ as promotionPrice,
        NVL(tPB.bundle_n, '') as bundleName,
        NVL(tPB.bundle_disp_seq, '1') as bundleDisplaySequence,
        tPB.sSelectYN as selectYN,
        tP.sPriceCompareServiceYN as priceCompareServiceYN,    
        NVL(tP.maker_c,0) as makerCode,
        NVL(tP.brand_c,0) as brandCode,    
        tP.movie_yn as movieYN,
        tP.sPriceLockYN as priceLockYN,
        NVL((select group_concat(nAttributeValueSeq, ',') from tProductAttributeValue tPA where tPA.nProductCategorySeq = tPC.nProductCategorySeq group by tPA.nProductCategorySeq) , '') as nAttributeValueSeq,
        NVL((select group_concat(sAttributeValueDescription, '|') from tProductAttributeValue tPA where tPA.nProductCategorySeq = tPC.nProductCategorySeq group by tPA.nProductCategorySeq) , '') as sAttributeValueDescription,
        tPDD.sSimpleDictionaryCode as simpleDictionaryCode,
        tPDD.sSummaryDictionaryCode as summaryDictionaryCode,
        NVL(tPC.nCntSave,0)+NVL(tSP.save_plus_q, 0) as savePlusQ,
        tUP.nTotalCapacity as totalCapacity,
        tUP.nStandardCapacity as standardCapacity,
        tUP.sUnit as unit,
        '1' as prodType,
        tCFS.sBRPS as brps,
        tPIP.nPriceType as PriceType,
        tPIP.nPrice as initialPrice,
        CASE
            WHEN tPIP.nPrice > 0 THEN
                TRUNC(((1-(tP.nLowPrice/tPIP.nPrice))*100),0)
            ELSE '' 
        END as discountRate,
        tP.disp_yn as dispYN,
        NVL(tPBC.nWriteCnt + tPBC.nCmpnyWriteCnt, 0) as writeCnt,
        tP.sProductRegisterType as productRegisterType,
        CASE
            WHEN tP.nLowPrice > 0 THEN 'N'
            WHEN tP.shot_yn = 'Y' THEN 'N'
            WHEN
            (
                tP.input_d >= DATEADD(SYSDATE, -30, 'DAY') 
                OR 
                tP.make_d >= DATEADD(SYSDATE, -30, 'DAY') 
                OR 
                TO_CHAR(tP.dtPriceModifyDate,'YYYYMMDD') >= TO_CHAR(DATEADD(SYSDATE, -30, 'DAY'),'YYYYMMDD')
            ) THEN 'N'
            ELSE 'Y'
        END as priceComparisonStopYN,
        tCFS.sDispYN as categoryDispYN,
        tPB.sSelectYN as bundleSelectYN,
        NVL((select group_concat(tRPTP.sModel,',') from tRelationProductTargetProduct tRPTP, tRelationProduct tRP where tRP.sDetailYN = 'Y' and tRPTP.nRelationProductSeq = tRP.nRelationProductSeq and tRP.nRelationFormValue = tP.prod_c group by tP.prod_c), '') as RELATIONMODEL,
        tPB.nDescriptionListSeq as descriptionListSeq,
        TO_CHAR(tP.modify_d , 'YYYYMMDDHHMISS') as imageVersion,
        NVL(tPBC.nCmpnyWriteScoreSum, 0) as cmpnyWriteScoreSum,
        NVL(tPBC.nCmpnyWriteCnt, 0) as cmpnyWriteCnt,
        tPEV.SVIDEOID as videoId,
        NVL(tPMP.NMAXPOPULARSCORE, 0) as maxPopularScore,
        TO_CHAR(NVL(tfds.FIRSTDATE, tp.MAKE_D), 'YYYYMMDD') as firstMakeDate,
        TO_CHAR(NVL(tfds.LASTDATE, tp.MAKE_D), 'YYYYMMDD') as lastMakeDate,
        NVL(tSS.NCATE2_SHARE,0) as shareCate2,
        NVL(tSS.NCATE3_SHARE,0) as shareCate3,
        NVL(tSS.NCATE4_SHARE,0) as shareCate4,
        NVL2(tm.NATTRIBUTEVALUESEQ, (tm.NATTRIBUTEVALUESEQ || '/' || tm.NCATEGORYSEQ || '.' || tm.SIMGTYPE), '') as optionMarkUrl,
        tm.NPOSITION as optionMarkPosition,
        NVL((select group_concat(TPS.KEYWORD,', ') from TPROD_SEARCH TPS WHERE TPS.PROD_C = tp.prod_c), '') AS searchKeyword,
        CASE 
            WHEN tP.avgprice_q > 0 THEN 
                CASE 
                    WHEN tP.nLowPrice/tP.avgprice_q > 0.5 THEN 'N'
                    ELSE 'Y'
                END
        END as priceDeviationYN,
        CASE
            WHEN tP.shot_yn = 'Y'  THEN 'D'
            ELSE    
                CASE
                    WHEN tP.sPriceCompareServiceYN = 'N' THEN
                        CASE
                            WHEN tP.make_d > SYSDATE THEN 'R'
                            ELSE 'B'
                        END
                    ELSE
                        CASE
                            WHEN tP.nLowPrice > 0 THEN 'N'
                            ELSE 
                                CASE
                                    WHEN TO_CHAR(tP.dtPriceModifyDate,'YYYYMMDD') >= TO_CHAR(DATEADD(SYSDATE, -30, 'DAY'),'YYYYMMDD') THEN 'O'
                                    ELSE
                                        CASE
                                            WHEN tP.dtPriceModifyDate != '01-JAN-1970' THEN 'S'
                                            ELSE
                                                CASE
                                                    WHEN tP.make_d > SYSDATE THEN 'R'
                                                    ELSE 
                                                        CASE
                                                            WHEN tP.input_d >= DATEADD(SYSDATE, -30,'DAY') THEN 'B'
                                                            ELSE 'S'
                                                        END
                                                END
                                        END
                                END
                        END
                END
        END as stateValue,
        NVL(tpsize.NWIDTH,0) as width,
        NVL(tpsize.NHEIGHT,0) as height,
        NVL(tpsize.NDEPTH,0) as depth,
        'N' as minabYN,
        CASE
            WHEN tP.SPRODUCTTYPE = 'A' THEN tP.PROD_C
            ELSE 0
        END as aiProductCode,
        tQDLP.CMPNY_C as quickShopCode,
        tQDLP.LINK_PROD_C as shopProductCode,
        NVL(tQDLP.NQUICK_DLVY_LOW_PRICE,0) as quickPrice,
        NVL(tulp.NUNIT_LOW_PRICE,0) as unitLowPrice,
        NVL(tulp.NUNIT_DLVY_LOW_PRICE,0) as unitTotalPrice,
        CASE
            WHEN NVL(tlp.NMEMBERSHIP_LOW_PRICE,0) = 0 THEN tP.nLowPrice
            ELSE tlp.NMEMBERSHIP_LOW_PRICE
        END as memberPrice,
        CASE
            WHEN NVL(tlp.NMEMBERSHIP_DLVY_LOW_PRICE,0) = 0 THEN tP.NLOWPRICEINCLUDEDELIVERYCOST
            ELSE tlp.NMEMBERSHIP_DLVY_LOW_PRICE
        END AS memberTotalPrice,
        CASE
            WHEN NVL(tlp.NMEMBERSHIP_UNIT_LOW_PRICE,0) = 0 THEN NVL(tulp.NUNIT_LOW_PRICE,0)
            ELSE tlp.NMEMBERSHIP_UNIT_LOW_PRICE
        END as unitMemberPrice,
        CASE
            WHEN NVL(tlp.NMEMBERSHIP_UNIT_DLVY_LOW_PRICE,0) = 0 THEN NVL(tulp.NUNIT_DLVY_LOW_PRICE,0)
            ELSE tlp.NMEMBERSHIP_UNIT_DLVY_LOW_PRICE
        END as unitMemberTotalPrice,
        CASE
            WHEN tep.NPRODUCTSEQ is not null AND tesp.NPRODUCTSEQ is not null THEN 'E'
            WHEN tep.NPRODUCTSEQ is null AND tesp.NPRODUCTSEQ is not null THEN 'ED'
            WHEN tep.NPRODUCTSEQ is null AND tesp.NPRODUCTSEQ is null THEN 'D'
            WHEN tep.NPRODUCTSEQ is not null AND tesp.NPRODUCTSEQ is null THEN 'ETC'
        END as enuriProductOption,
        CASE
            WHEN NVL(tlp.NMEMBERSHIP_LOW_PRICE, tP.nLowPrice + 1) <= tP.nLowPrice THEN 'Y'
            ELSE 'N'
        END as memberPriceLowerYN,
        NVL(TO_CHAR(tvml.DTLAST_MATCH_DATE, 'YYYYMMDD'),0) as vectorMatchDate,
        tci.IMAGE_PATH as imagePath
    FROM
        tprod tP
        INNER JOIN tProductCategory tPC ON (tP.prod_c = tPC.nProductSeq)
        INNER JOIN tCategoryForSearch tCFS ON (tPC.nCategorySeq1 = tCFS.nCategorySeq1
            and tPC.nCategorySeq2 = tCFS.nCategorySeq2
            and tPC.nCategorySeq3 = tCFS.nCategorySeq3
            and tPC.nCategorySeq4 = tCFS.nCategorySeq4)
        INNER JOIN tProductDescription tPD ON (tPC.nProductCategorySeq = tPD.nProductCategorySeq)
        INNER JOIN tprod_maker tPM ON (tP.maker_c = tPM.maker_c)
        LEFT JOIN tProductCategoryDescriptionActionTag tPCDAT ON (tPC.nProductCategorySeq = tPCDAT.nProductCategorySeq)
        LEFT JOIN tprod_brand tPR ON (tP.brand_c = tPR.brand_c)
        LEFT JOIN tprod_bundle AS tPB ON (tP.prod_c = tPB.prod_c)
        LEFT JOIN tKeywordForSearch tKFS ON (tP.prod_c = tKFS.nProductSeq)
        LEFT JOIN tsave_plus tSP ON (tP.prod_c = tSP.prod_c)
        LEFT JOIN tGoodsUnitPrice tUP ON (tUP.nGoodsCode = tP.prod_c)
        LEFT JOIN tProductDescriptionDictionary tPDD ON (tPD.nProductCategorySeq = tPDD.nProductCategorySeq)
        LEFT JOIN tProductInitialPrice tPIP ON (tP.prod_c = tPIP.nProductSeq)
        LEFT JOIN tProductBoardCount tPBC ON (tP.prod_c = tPBC.nProductCode)
        LEFT JOIN tProd_Search_Manage tPSM ON (tP.prod_c = tPSM.prod_c)
        LEFT JOIN TPRODUCTEXTERNALVIDEO tPEV ON (tP.prod_c = tPEV.NPRODUCTSEQ and tPEV.NVIDEONUMBER=1)
        LEFT JOIN TPRODMAXPOPULARSCORE tPMP ON (tP.prod_c = tPMP.NPRODUCTSEQ)
        LEFT JOIN TFIRSTDATEFORSEARCH tfds ON (tP.prod_c = tfds.prod_c)
        LEFT JOIN TPRODSALESSHARE tSS ON (tP.prod_c = tSS.NPRODUCTSEQ)
        LEFT JOIN TMARKPRODUCT tmp ON (tP.prod_c = tmp.NPRODUCTSEQ)
        LEFT JOIN TMARK tm ON (tmp.NATTRIBUTEVALUESEQ = tm.NATTRIBUTEVALUESEQ and tmp.NCATEGORYSEQ = tm.NCATEGORYSEQ)
        LEFT JOIN TPRODSIZE tpsize ON (tP.prod_c = tpsize.NPRODCODE)
        LEFT JOIN TQUICK_DLVY_LOW_PRICE_LINK tQDLP ON (tp.PROD_C = tQDLP.PROD_C)
        LEFT JOIN TENURI_PRODUCT tep ON (tp.PROD_C = tep.NPRODUCTSEQ)
        LEFT JOIN TUNIT_LOW_PRICE tulp ON (tp.PROD_C = tulp.PROD_C)
        LEFT JOIN TMEMBERSHIP_LOW_PRICE tlp ON (tp.PROD_c = tlp.PROD_C)
        LEFT JOIN TENURI_SOURCE_PRODUCT tesp ON (tp.PROD_C = tesp.NPRODUCTSEQ)
        LEFT JOIN TPROD_VECTOR_MATCH_LOG tvml ON (tP.prod_c = tvml.PROD_C)
        LEFT JOIN TCATALOGIMAGE tci ON (tp.PROD_C = tci.PROD_C)
    WHERE
        NOT tPC.nCategorySeq2 =19563
